# kbXplorer — Python 3.12, CPU only.

# Versions are lower bounds pinned to the majors this project was built and
# tested against.

tokenizers>=0.19          # tokenisation for the ONNX embedder

# --- Local LLM -------------------------------------------------------------
ollama>=0.3               # client for the local Ollama daemon

# --- Interface -------------------------------------------------------------
textual>=0.80             # dual-pane TUI framework
rich>=13                  # text rendering (also a Textual dependency)

# --- Folder monitoring -----------------------------------------------------
watchdog>=4               # filesystem events for automatic re-indexing

# --- Imaging ---------------------------------------------------------------
pillow>=10                # figure post-processing and size checks
